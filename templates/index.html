Ryoku API

Welcome to the official documentation for the Ryoku API. Ryoku is a world-class, polyglot AI tutor, an expert in all human and programming languages. This API allows you to integrate Ryoku's teaching capabilities into your own applications.

Capabilities

  • Polyglot Expertise: Teaches any programming language (Python, JavaScript, etc.) and any human language (Spanish, Japanese, etc.).
  • Adaptive Teaching: Automatically adapts its style for programming concepts versus linguistic nuances.
  • Context-Aware: Remembers the conversation history to provide coherent and relevant lessons (feature in development).
  • Simple Integration: A straightforward POST endpoint for easy integration.

API Usage

To use the API, send a POST request to the /chat endpoint with a JSON body containing your user_id and new_message.


import requests
import json

API_URL = "https://ryoku-tutor-api.onrender.com/chat"

def ask_ryoku(message, user_id="demo-user"):
    payload = {
        "user_id": user_id,
        "new_message": message
    }
    headers = {"Content-Type": "application/json"}

    try:
        response = requests.post(API_URL, data=json.dumps(payload), headers=headers)
        response.raise_for_status()
        print(response.json()["answer"])
    except requests.exceptions.RequestException as e:
        print(f"Error: {e}")

ask_ryoku("Explain what a variable is in Python.")


const API_URL = "https://ryoku-tutor-api.onrender.com/chat";

async function askRyoku(message, userId = "demo-user") {
    const payload = {
        user_id: userId,
        new_message: message
    };

    try {
        const response = await fetch(API_URL, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
        });

        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }

        const data = await response.json();
        console.log(data.answer);
    } catch (error) {
        console.error("Error:", error);
    }
}

// Requires a library like OkHttp or Java 11's HttpClient
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class RyokuClient {
    public static void main(String[] args) throws Exception {
        String apiUrl = "https://ryoku-tutor-api.onrender.com/chat";
        String jsonPayload = "{\"user_id\": \"demo-user\", \"new_message\": \"What is the main method in Java?\"}";

        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(apiUrl))
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(jsonPayload))
                .build();

        HttpResponse response = client.send(request, HttpResponse.BodyHandlers.ofString());
        System.out.println(response.body());
    }
}


// Requires HttpClient, available in .NET
using System;
using System.Net.Http;
using System.Text;
using System.Threading.Tasks;

public class RyokuClient
{
    static async Task Main(string[] args)
    {
        var apiUrl = "https://ryoku-tutor-api.onrender.com/chat";
        var client = new HttpClient();

        var payload = new {
            user_id = "demo-user",
            new_message = "What is LINQ in C#?"
        };

        var jsonPayload = System.Text.Json.JsonSerializer.Serialize(payload);
        var content = new StringContent(jsonPayload, Encoding.UTF8, "application/json");

        try {
            HttpResponseMessage response = await client.PostAsync(apiUrl, content);
            response.EnsureSuccessStatusCode();
            string responseBody = await response.Content.ReadAsStringAsync();
            Console.WriteLine(responseBody);
        } catch (HttpRequestException e) {
            Console.WriteLine($"Error: {e.Message}");
        }
    }
}


<?php
$apiUrl = 'https://ryoku-tutor-api.onrender.com/chat';

function askRyoku($message, $userId = 'demo-user') {
    global $apiUrl;

    $payload = json_encode([
        'user_id' => $userId,
        'new_message' => $message
    ]);

    $ch = curl_init($apiUrl);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        'Content-Type: application/json',
        'Content-Length: ' . strlen($payload)
    ]);

    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    if ($httpCode >= 200 && $httpCode < 300) {
        echo $response;
    } else {
        echo "Error: Received status code " . $httpCode;
    }
}

askRyoku("What is an associative array in PHP?");
